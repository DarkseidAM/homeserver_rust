13/// Times we skipped a per-interface rate sample because a cumulative counter decreased (reset / driver quirk).
14static NETWORK_RATE_COUNTER_DECREASE_SKIPS: AtomicU64 = AtomicU64::new(0);
15
16pub(super) fn log_counter_decrease(iface: &str, field: &str, curr: u64, prev: u64) {
17    let n = NETWORK_RATE_COUNTER_DECREASE_SKIPS.fetch_add(1, Ordering::Relaxed) + 1;
18    tracing::debug!(
19        operation = "network_rate_skip",
20        iface = %iface,
21        field = field,
22        curr = curr,
23        prev = prev,
24        skips_total = n,
25        "cumulative {} counter decreased; skipping rate this interval",
26        field
27    );
28}

30pub(super) fn collect_storage_inner(
31    disks: &Mutex<Disks>,
32    disk_models: &Mutex<HashMap<String, String>>,
33) -> StorageStats {
34    let mut disks_guard = match disks.lock() {
35        Ok(g) => g,
36        Err(e) => {
37            tracing::warn!(error = %e, "sysinfo disks lock poisoned");
38            return StorageStats::default();
39        }
40    };
41
42    disks_guard.refresh(true);
43    let disk_count = disks_guard.list().len();
44    let mut partitions: Vec<PartitionStat> = Vec::with_capacity(disk_count);
45    for d in disks_guard.list() {
46        let total = d.total_space();
47        let available = d.available_space();
48        let used = total.saturating_sub(available);
49        let usage_percent = if total > 0 {
50            (used as f64 / total as f64) * 100.0
51        } else {
52            0.0
53        };
54        partitions.push(PartitionStat {
55            mount: d.mount_point().to_string_lossy().into_owned(),
56            name: d.name().to_string_lossy().into_owned(),
57            type_: d.file_system().to_string_lossy().into_owned(),
58            total_space: total,
59            used_space: used,
60            available_space: available,
61            usage_percent,
62        });
63    }
64
65    let diskstats = linux::read_diskstats_linux();
66    let mut disk_devices: Vec<DiskDeviceStat> = Vec::with_capacity(disk_count);
67    let mut cached_models = disk_models.lock().ok();
68
69    for d in disks_guard.list() {
70        let raw_name: String = d.name().to_string_lossy().into_owned();
71        let dev_name = raw_name.trim_start_matches("/dev/").to_string();
72        let io = diskstats.get(&dev_name).cloned().unwrap_or_default();
73        let model = if let Some(ref mut map) = cached_models {
74            if let Some(m) = map.get(&dev_name) {
75                m.clone()
76            } else {
77                let m = linux::read_disk_model_linux(&dev_name);
78                map.insert(dev_name.clone(), m.clone());
79                m
80            }
81        } else {
82            linux::read_disk_model_linux(&dev_name)
83        };
84        disk_devices.push(DiskDeviceStat {
85            name: raw_name,
86            model,
87            size: d.total_space(),
88            read_bytes: io.sectors_read * 512,
89            write_bytes: io.sectors_written * 512,
90            io_time_ms: io.io_time_ms,
91            iops_read: io.reads_completed,
92            iops_write: io.writes_completed,
93        });
94    }
95
96    StorageStats {
97        partitions,
98        disks: disk_devices,
99    }
100}

102pub(super) fn collect_network_inner(
103    networks: &Mutex<Networks>,
104    last_network: &Mutex<Option<(HashMap<String, NetworkCounters>, Instant)>>,
105    iface_speeds: &Mutex<HashMap<String, u64>>,
106) -> NetworkStats {
107    let mut networks_guard = match networks.lock() {
108        Ok(g) => g,
109        Err(e) => {
110            tracing::warn!(error = %e, "sysinfo networks lock poisoned");
111            return NetworkStats::default();
112        }
113    };
114
115    networks_guard.refresh(true);
116    let iface_count = networks_guard.list().len();
117    let mut interfaces: Vec<InterfaceStat> = Vec::with_capacity(iface_count);
118    let mut cached_speeds = iface_speeds.lock().ok();
119
120    for (name, data) in networks_guard.list() {
121        let speed = if let Some(ref mut map) = cached_speeds {
122            if let Some(&s) = map.get(name) {
123                s
124            } else {
125                let s = linux::get_interface_speed(name);
126                map.insert(name.clone(), s);
127                s
128            }
129        } else {
130            linux::get_interface_speed(name)
131        };
132
133        interfaces.push(InterfaceStat {
134            name: name.clone(),
135            display_name: name.clone(),
136            mac_address: data.mac_address().to_string(),
137            ipv4: data
138                .ip_networks()
139                .iter()
140                .filter(|n| n.addr.is_ipv4())
141                .map(|n| n.addr.to_string())
142                .collect(),
143            ipv6: data
144                .ip_networks()
145                .iter()
146                .filter(|n| n.addr.is_ipv6())
147                .map(|n| n.addr.to_string())
148                .collect(),
149            bytes_sent: data.transmitted(),
150            bytes_recv: data.received(),
151            packets_sent: data.packets_transmitted(),
152            packets_recv: data.packets_received(),
153            speed,
154            received_bytes_per_sec: 0.0,
155            transmitted_bytes_per_sec: 0.0,
156            is_up: linux::read_interface_operstate(name),
157        });
158    }
159
160    let now = Instant::now();
161    if let Ok(mut last_guard) = last_network.lock() {
162        if let Some((ref prev_map, prev_ts)) = *last_guard {
163            let dt_secs = now.duration_since(prev_ts).as_secs_f64();
164            if dt_secs > 0.0 {
165                for iface in &mut interfaces {
166                    if let Some(p) = prev_map.get(&iface.name) {
167                        if iface.bytes_recv >= p.bytes_recv {
168                            let drx = iface.bytes_recv - p.bytes_recv;
169                            iface.received_bytes_per_sec = drx as f64 / dt_secs;
170                        } else {
171                            log_counter_decrease(
172                                &iface.name,
173                                "bytes_recv",
174                                iface.bytes_recv,
175                                p.bytes_recv,
176                            );
177                        }
178                        if iface.bytes_sent >= p.bytes_sent {
179                            let dtx = iface.bytes_sent - p.bytes_sent;
180                            iface.transmitted_bytes_per_sec = dtx as f64 / dt_secs;
181                        } else {
182                            log_counter_decrease(
183                                &iface.name,
184                                "bytes_sent",
185                                iface.bytes_sent,
186                                p.bytes_sent,
187                            );
188                        }
189                    }
190                }
191            }
192        }
193        let mut new_map = HashMap::with_capacity(interfaces.len());
194        for iface in &interfaces {
195            new_map.insert(
196                iface.name.clone(),
197                NetworkCounters {
198                    bytes_recv: iface.bytes_recv,
199                    bytes_sent: iface.bytes_sent,
200                },
201            );
202        }
203        *last_guard = Some((new_map, now));
204    }
205
206    NetworkStats { interfaces }
207}

209impl SysinfoRepo {
210    #[instrument(skip(self), fields(repo = "sysinfo", operation = "get_storage_stats"))]
211    pub async fn get_storage_stats(&self) -> anyhow::Result<StorageStats> {
212        let disks = self.disks.clone();
213        let disk_models = self.disk_models.clone();
214        tokio::task::spawn_blocking(move || Ok(collect_storage_inner(&disks, &disk_models)))
215            .await
216            .map_err(|e| anyhow::anyhow!("sysinfo task join: {}", e))?
217    }
218
219    #[instrument(skip(self), fields(repo = "sysinfo", operation = "get_network_stats"))]
220    pub async fn get_network_stats(&self) -> anyhow::Result<NetworkStats> {
221        let networks = self.networks.clone();
222        let last_network = self.last_network.clone();
223        let iface_speeds = self.iface_speeds.clone();
224        tokio::task::spawn_blocking(move || {
225            Ok(collect_network_inner(
226                &networks,
227                &last_network,
228                &iface_speeds,
229            ))
230        })
231        .await
232        .map_err(|e| anyhow::anyhow!("sysinfo task join: {}", e))?
233    }
